## 1. Copy this code into Thonny
 
 
```python
 
from dataclasses import dataclass
 
from drafter import *
 
 
 
# Sample information for Campus Connect
 
STUDENTS = ["Alex", "Sam", "Jordan"]
 
DAYS = ["Monday", "Tuesday", "Wednesday", "Thursday", "Friday"]
 
TIMES = ["10 AM", "12 PM", "2 PM", "4 PM", "6 PM"]
 
ACTIVITIES = ["Coffee", "Study", "Gym", "Food"]
 
 
 
@dataclass
 
class ActivityInvitation:
 
    sender: str
 
    receiver: str
 
    activity: str
 
    day: str
 
    time: str
 
 
 
@dataclass
 
class State:
 
    session: str
 
    selected_day: str
 
    available_slots: dict[str, list[str]]
 
    invitations: list[ActivityInvitation]
 
 
 
def menu() -> list:
 
    """Buttons that appear on every page."""
 
    return [
 
        Link("Home", "index"),
 
        " | ",
 
        Link("Calendar", "calendar"),
 
        " | ",
 
        Link("Friends", "friends"),
 
        " | ",
 
        Link("Suggest Activity", "suggest"),
 
        " | ",
 
        Link("Inbox", "inbox"),
 
        "\n\n",
 
    ]
 
 
 
@route
 
def index(state: State) -> Page:
 
    return Page(state, [
 
        Header("Campus Connect"),
 
        "Find free time with your friends and make plans!\n",
 
        "Signed in as: " + state.session + "\n\n",
 
        "Choose a student:",
 
        SelectBox("student", STUDENTS, state.session),
 
        Button("Sign In", "sign_in"),
 
        "\n\n",
 
    ] + menu())
 
 
 
@route
 
def sign_in(state: State, student: str) -> Page:
 
    state.session = student
 
    return index(state)
 
 
 
@route
 
def calendar(state: State) -> Page:
 
    saved = state.available_slots[state.session]
 
    lines = [
 
        Header("My Calendar"),
 
        "Student: " + state.session + "\n",
 
        "Choose a day:",
 
        SelectBox("day", DAYS, state.selected_day),
 
        "\n Choose a time:",
 
        SelectBox("time", TIMES),
 
        "\n",
 
        Button("Add Free Time", "save_availability"),
 
        Button("Remove Free Time", "remove_availability"),
 
        "\n\nMy free times:\n",
 
    ]
 
 
    if len(saved) == 0:
 
        lines.append("No free times added yet.\n")
 
    else:
 
        for slot in saved:
 
            lines.append(slot + "\n")
 
 
    return Page(state, menu() + lines)
 
 
 
@route
 
def save_availability(state: State, day: str, time: str) -> Page:
 
    state.selected_day = day
 
    slot = day + " at " + time
 
 
    if slot not in state.available_slots[state.session]:
 
        state.available_slots[state.session].append(slot)
 
 
    return calendar(state)
 
 
 
@route
 
def remove_availability(state: State, day: str, time: str) -> Page:
 
    state.selected_day = day
 
    slot = day + " at " + time
 
 
    if slot in state.available_slots[state.session]:
 
        state.available_slots[state.session].remove(slot)
 
 
    return calendar(state)
 
 
 
@route
 
def friends(state: State) -> Page:
 
    lines = [Header("Friends"), "Your friends' free times:\n\n"]
 
 
    for student in STUDENTS:
 
        if student != state.session:
 
            lines.append(Header(student, 3))
 
            slots = state.available_slots[student]
 
 
            if len(slots) == 0:
 
                lines.append("No free times added yet.\n")
 
            else:
 
                for slot in slots:
 
                    lines.append(slot + "\n")
 
 
    return Page(state, menu() + lines)
 
 
 
@route
 
def suggest(state: State) -> Page:
 
    other_students = []
 
 
    for student in STUDENTS:
 
        if student != state.session:
 
            other_students.append(student)
 
 
    return Page(state, menu() + [
 
        Header("Suggest Activity"),
 
        "Invite a friend to do something together.\n\n",
 
        "Friend:",
 
        SelectBox("friend", other_students),
 
        "\n Activity:",
 
        SelectBox("activity", ACTIVITIES),
 
        "\n Day:",
 
        SelectBox("day", DAYS, state.selected_day),
 
        "\n Time:",
 
        SelectBox("time", TIMES),
 
        "\n",
 
        Button("Send Invitation", "send_suggestion"),
 
    ])
 
 
 
@route
 
def send_suggestion(state: State, friend: str, activity: str,
 
                    day: str, time: str) -> Page:
 
    slot = day + " at " + time
 
 
    if slot not in state.available_slots[state.session]:
 
        message = "You need to mark yourself free at that time first."
 
 
    elif slot not in state.available_slots[friend]:
 
        message = friend + " has not marked that time as free yet."
 
 
    else:
 
        invitation = ActivityInvitation(
 
            state.session, friend, activity, day, time
 
        )
 
        state.invitations.append(invitation)
 
        message = "Invitation sent to " + friend + " for " + activity + "!"
 
 
    return Page(state, menu() + [
 
        Header("Invitation Result"),
 
        message,
 
        "\n",
 
        Button("Try Another Activity", "suggest"),
 
    ])
 
 
 
@route
 
def inbox(state: State) -> Page:
 
    lines = [
 
        Header("Inbox"),
 
        "Invitations for " + state.session + ":\n\n"
 
    ]
 
    found = False
 
 
    # Loop through a list of ActivityInvitation objects
 
    for invitation in state.invitations:
 
        if invitation.receiver == state.session:
 
            found = True
 
            lines.append(
 
                invitation.sender + " invited you to " +
 
                invitation.activity + " on " + invitation.day +
 
                " at " + invitation.time + ".\n"
 
            )
 
 
    if not found:
 
        lines.append("You have no invitations yet.\n")
 
 
    return Page(state, menu() + lines)
 
 
 
# Start the website
 
if __name__ == "__main__":
 
    start_server(State(
 
        "Alex",
 
        "Monday",
 
        {"Alex": [], "Sam": [], "Jordan": []},
 
        []
 
    ))
 
```
